function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Sheet1") || SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; var rows = sheet.getDataRange().getValues(); var headers = rows[0]; var data = []; for (var i = 1; i < rows.length; i++) { var obj = {}; for (var j = 0; j < headers.length; j++) { obj[headers[j]] = rows[i][j]; } data.push(obj); } var output = ContentService.createTextOutput(JSON.stringify(data)); output.setMimeType(ContentService.MimeType.JSON); return output; } function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Sheet1") || SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; try { var data = JSON.parse(e.postData.contents); var rows = sheet.getDataRange().getValues(); var headers = rows[0]; var idCol = headers.indexOf("Part ID"); var qtyCol = headers.indexOf("Current Stock Qty"); // 1. Agar request Bill Save karne ki hai (Reports / Invoices Archive) if (data.action === "save_bill") { // Sheet me Headers check karlo, agar Column H (index 7) tak headers nahi hain toh set kar sakte hain sheet.appendRow([ data.invoiceId, data.custName, data.custMobile, data.vehicleNo, data.paymentMode, data.grandTotal, data.date, JSON.stringify(data) // Column H me poora bill JSON save ho jayega entry-wise ]); var output = ContentService.createTextOutput(JSON.stringify({status: "success", message: "Bill saved to sheet"})); output.setMimeType(ContentService.MimeType.JSON); return output; } // 2. Agar stock update karne ki request hai var foundRow = -1; if (data.id && idCol !== -1) { for (var i = 1; i < rows.length; i++) { if (String(rows[i][idCol]) === String(data.id)) { foundRow = i + 1; break; } } } if (foundRow !== -1 && data.action === "update_stock") { sheet.getRange(foundRow, qtyCol + 1).setValue(data.qty); var output = ContentService.createTextOutput(JSON.stringify({status: "success", message: "Stock updated"})); output.setMimeType(ContentService.MimeType.JSON); return output; } // 3. Naya inventory part add karne ke liye sheet.appendRow([data.id, data.name, data.hsn, data.pPrice, data.sPrice, data.qty, new Date()]); var output = ContentService.createTextOutput(JSON.stringify({status: "success", message: "Row appended"})); output.setMimeType(ContentService.MimeType.JSON); return output; } catch(err) { var output = ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()})); output.setMimeType(ContentService.MimeType.JSON); return output; } }
